     1→'use client'
     2→
     3→import { useEffect, useMemo, useState } from 'react'
     4→import { apiGet, apiSend, fmtDate, dateInput, todayInput } from '@/lib/api'
     5→import { downloadInspectionSheet, downloadUnitHistoryReport } from '@/lib/pdf'
     6→import type { Unit, UnitEquipment, InspectionTemplate, AirconModel, Inspection } from '@/lib/types'
     7→import { ViewTitle, StatusBadge, DueBadge } from '@/components/views/shared'
     8→import { InspectionForm } from '@/components/inspection-form'
     9→import { Button } from '@/components/ui/button'
    10→import { Input } from '@/components/ui/input'
    11→import { Label } from '@/components/ui/label'
    12→import { Badge } from '@/components/ui/badge'
    13→import { Card, CardContent } from '@/components/ui/card'
    14→import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog'
    15→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    16→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    17→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table'
    18→import { AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle } from '@/components/ui/alert-dialog'
    19→import { toast } from 'sonner'
    20→import { Boxes, Plus, Search, Pencil, Trash2, Eye, History as HistoryIcon, ArrowUpRight, ArrowDownLeft, FileDown, Wrench, Package } from 'lucide-react'
    21→
    22→const emptyUnit = {
    23→  id: '', code: '', serial: '', type: 'CONTAINER', size: '', description: '', status: 'IN_STOCK',
    24→  client: '', site: '', dueBack: '', buildDate: '',
    25→  builder: '', electrician: '', plumber: '', painter: '', glazier: '', roofer: '', otherTrade: '', notes: '',
    26→}
    27→
    28→export function UnitsView({ refreshKey, onNavigate }: { refreshKey: number; onNavigate: (v: string) => void }) {
    29→  const [units, setUnits] = useState<Unit[]>([])
    30→  const [templates, setTemplates] = useState<InspectionTemplate[]>([])
    31→  const [aircons, setAircons] = useState<AirconModel[]>([])
    32→  const [loading, setLoading] = useState(true)
    33→  const [q, setQ] = useState('')
    34→  const [statusFilter, setStatusFilter] = useState('ALL')
    35→  const [editOpen, setEditOpen] = useState(false)
    36→  const [form, setForm] = useState<typeof emptyUnit>(emptyUnit)
    37→  const [detail, setDetail] = useState<Unit | null>(null)
    38→  const [detailTab, setDetailTab] = useState('overview')
    39→  const [deleteId, setDeleteId] = useState<string | null>(null)
    40→  const [inspOpen, setInspOpen] = useState(false)
    41→  const [inspType, setInspType] = useState<'OUTGOING' | 'RETURN' | 'QCP'>('OUTGOING')
    42→  const [inspUnitId, setInspUnitId] = useState('')
    43→  const [eqEdit, setEqEdit] = useState<Partial<UnitEquipment> | null>(null)
    44→  const [generating, setGenerating] = useState(false)
    45→
    46→  const load = async () => {
    47→    setLoading(true)
    48→    try {
    49→      const [u, t, a] = await Promise.all([
    50→        apiGet<Unit[]>('/api/units'),
    51→        apiGet<InspectionTemplate[]>('/api/qcp'),
    52→        apiGet<AirconModel[]>('/api/aircons'),
    53→      ])
    54→      setUnits(u); setTemplates(t); setAircons(a)
    55→    } catch (e) { toast.error(e.message) } finally { setLoading(false) }
    56→  }
    57→  useEffect(() => { load() }, [refreshKey])
    58→
    59→  const filtered = useMemo(() => units.filter((u) => {
    60→    const hay = `${u.code} ${u.description} ${u.client} ${u.site} ${u.serial}`.toLowerCase()
    61→    return (statusFilter === 'ALL' || u.status === statusFilter) && hay.includes(q.toLowerCase())
    62→  }), [units, q, statusFilter])
    63→
    64→  const saveUnit = async () => {
    65→    if (!form.code.trim()) { toast.error('Unit code is required'); return }
    66→    try {
    67→      if (form.id) await apiSend('/api/units', 'PUT', form)
    68→      else await apiSend('/api/units', 'POST', form)
    69→      toast.success(form.id ? 'Unit updated' : 'Unit added')
    70→      setEditOpen(false)
    71→      load()
    72→    } catch (e) { toast.error(e.message) }
    73→  }
    74→
    75→  const del = async () => {
    76→    if (!deleteId) return
    77→    try { await apiSend(`/api/units?id=${deleteId}`, 'DELETE'); toast.success('Unit deleted'); setDeleteId(null); load() }
    78→    catch (e) { toast.error(e.message) }
    79→  }
    80→
    81→  const generateSchedules = async (unit: Unit) => {
    82→    setGenerating(true)
    83→    try {
    84→      const r = await apiSend<{ created: number }>('/api/maintenance', 'POST', { action: 'generate_for_unit', unitId: unit.id })
    85→      toast.success(`Created ${r.created} maintenance schedule(s) for ${unit.code}`)
    86→      load()
    87→      // refresh detail
    88→      const fresh = await apiGet<Unit[]>('/api/units')
    89→      setDetail(fresh.find((x) => x.id === unit.id) || null)
    90→    } catch (e) { toast.error(e.message) } finally { setGenerating(false) }
    91→  }
    92→
    93→  const saveEquipment = async () => {
    94→    if (!eqEdit || !detail) return
    95→    if (!eqEdit.name?.trim()) { toast.error('Item name required'); return }
    96→    try {
    97→      if (eqEdit.id) await apiSend('/api/units', 'PUT', { ...eqEdit }) // not used – equipment has own API? keep below
    98→    } catch { /* noop */ }
    99→    // Equipment is managed via dedicated endpoint below
   100→    try {
   101→      const payload = { ...eqEdit, unitId: detail.id, qty: Number(eqEdit.qty) || 1, installedDate: eqEdit.installedDate || null }
   102→      if (eqEdit.id) await apiSend('/api/equipment', 'PUT', payload)
   103→      else await apiSend('/api/equipment', 'POST', payload)
   104→      toast.success('Contents saved')
   105→      setEqEdit(null)
   106→      const fresh = await apiGet<Unit[]>('/api/units')
   107→      setUnits(fresh)
   108→      setDetail(fresh.find((x) => x.id === detail.id) || null)
   109→    } catch (e) { toast.error(e.message) }
   110→  }
   111→
   112→  const delEquipment = async (id: string) => {
   113→    try {
   114→      await apiSend(`/api/equipment?id=${id}`, 'DELETE')
   115→      const fresh = await apiGet<Unit[]>('/api/units')
   116→      setUnits(fresh)
   117→      setDetail(fresh.find((x) => x.id === detail!.id) || null)
   118→      toast.success('Item removed')
   119→    } catch (e) { toast.error(e.message) }
   120→  }
   121→
   122→  const nextDue = (u: Unit) => {
   123→    const dues = (u.schedules || []).map((s) => s.nextDue).filter(Boolean) as string[]
   124→    return dues.length ? dues.sort()[0] : null
   125→  }
   126→
   127→  const openInsp = (unit: Unit, type: 'OUTGOING' | 'RETURN' | 'QCP') => {
   128→    setInspUnitId(unit.id); setInspType(type); setInspOpen(true)
   129→  }
   130→
   131→  const downloadInsp = (insp: Inspection, unit?: Unit) => {
   132→    downloadInspectionSheet(
   133→      { ...insp, unit: unit ? { code: unit.code, description: unit.description } : insp.unit } as never,
   134→      'Maintenance',
   135→      unit,
   136→    )
   137→  }
   138→
   139→  return (
   140→    <div>
   141→      <ViewTitle
   142→        title="Units register & stock take"
   143→        subtitle="Every container / prefab unit, its build team, contents and history"
   144→        action={<Button size="sm" onClick={() => { setForm({ ...emptyUnit }); setEditOpen(true) }}><Plus className="h-4 w-4 mr-1" /> Add unit</Button>}
   145→      />
   146→
   147→      <div className="flex flex-wrap gap-2 mb-4">
   148→        <div className="relative flex-1 min-w-52">
   149→          <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   150→          <Input className="pl-8" placeholder="Search code, client, serial…" value={q} onChange={(e) => setQ(e.target.value)} />
   151→        </div>
   152→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   153→          <SelectTrigger className="w-44"><SelectValue /></SelectTrigger>
   154→          <SelectContent>
   155→            <SelectItem value="ALL">All statuses</SelectItem>
   156→            <SelectItem value="IN_STOCK">In stock</SelectItem>
   157→            <SelectItem value="CLIENT_SITE">At client</SelectItem>
   158→            <SelectItem value="MAINTENANCE">In maintenance</SelectItem>
   159→            <SelectItem value="RETIRED">Retired</SelectItem>
   160→          </SelectContent>
   161→        </Select>
   162→      </div>
   163→
   164→      <Card className="mb-4">
   165→        <CardContent className="px-2 sm:px-4 py-0 overflow-x-auto">
   166→          <Table>
   167→            <TableHeader>
   168→              <TableRow>
   169→                <TableHead>Code</TableHead>
   170→                <TableHead className="hidden sm:table-cell">Type</TableHead>
   171→                <TableHead>Description</TableHead>
   172→                <TableHead className="hidden md:table-cell">Status</TableHead>
   173→                <TableHead className="hidden lg:table-cell">Client</TableHead>
   174→                <TableHead>Next due</TableHead>
   175→                <TableHead className="text-right">Actions</TableHead>
   176→              </TableRow>
   177→            </TableHeader>
   178→            <TableBody>
   179→              {loading ? (
   180→                <TableRow><TableCell colSpan={7} className="text-center py-8 text-muted-foreground">Loading…</TableCell></TableRow>
   181→              ) : filtered.length === 0 ? (
   182→                <TableRow><TableCell colSpan={7} className="text-center py-8 text-muted-foreground">No units found.</TableCell></TableRow>
   183→              ) : filtered.map((u) => (
   184→                <TableRow key={u.id}>
   185→                  <TableCell className="font-medium">{u.code}</TableCell>
   186→                  <TableCell className="hidden sm:table-cell text-sm">{u.type} {u.size}</TableCell>
   187→                  <TableCell className="text-sm max-w-56 truncate">{u.description}</TableCell>
   188→                  <TableCell className="hidden md:table-cell"><StatusBadge status={u.status} /></TableCell>
   189→                  <TableCell className="hidden lg:table-cell text-sm">{u.client || '-'}</TableCell>
   190→                  <TableCell><div className="flex items-center gap-1.5">{nextDue(u) && <DueBadge date={nextDue(u)} />}<span className="text-xs text-muted-foreground">{fmtDate(nextDue(u))}</span></div></TableCell>
   191→                  <TableCell className="text-right">
   192→                    <div className="flex justify-end gap-1">
   193→                      <Button variant="ghost" size="icon" className="h-8 w-8" title="View" onClick={() => { setDetail(u); setDetailTab('overview') }}><Eye className="h-4 w-4" /></Button>
   194→                      {u.status !== 'CLIENT_SITE' && (
   195→                        <Button variant="ghost" size="icon" className="h-8 w-8 text-emerald-700" title="Sign out to client" onClick={() => openInsp(u, 'OUTGOING')}><ArrowUpRight className="h-4 w-4" /></Button>
   196→                      )}
   197→                      {u.status === 'CLIENT_SITE' && (
   198→                        <Button variant="ghost" size="icon" className="h-8 w-8 text-amber-700" title="Receive back from client" onClick={() => openInsp(u, 'RETURN')}><ArrowDownLeft className="h-4 w-4" /></Button>
   199→                      )}
   200→                      <Button variant="ghost" size="icon" className="h-8 w-8" title="Edit" onClick={() => { setForm({ ...emptyUnit, ...u, buildDate: dateInput(u.buildDate), dueBack: dateInput(u.dueBack) } as typeof emptyUnit); setEditOpen(true) }}><Pencil className="h-4 w-4" /></Button>
   201→                      <Button variant="ghost" size="icon" className="h-8 w-8 text-red-600" title="Delete" onClick={() => setDeleteId(u.id)}><Trash2 className="h-4 w-4" /></Button>
   202→                    </div>
   203→                  </TableCell>
   204→                </TableRow>
   205→              ))}
   206→            </TableBody>
   207→          </Table>
   208→        </CardContent>
   209→      </Card>
   210→
   211→      {/* Add / edit unit */}
   212→      <Dialog open={editOpen} onOpenChange={setEditOpen}>
   213→        <DialogContent className="max-w-2xl max-h-[92vh] overflow-y-auto">
   214→          <DialogHeader><DialogTitle>{form.id ? `Edit unit ${form.code}` : 'Add unit'}</DialogTitle></DialogHeader>
   215→          <div className="grid gap-3 sm:grid-cols-2">
   216→            <div><Label>Unit code *</Label><Input value={form.code} onChange={(e) => setForm({ ...form, code: e.target.value })} placeholder="e.g. PC-006" /></div>
   217→            <div><Label>Serial no.</Label><Input value={form.serial} onChange={(e) => setForm({ ...form, serial: e.target.value })} /></div>
   218→            <div>
   219→              <Label>Type</Label>
   220→              <Select value={form.type} onValueChange={(v) => setForm({ ...form, type: v })}>
   221→                <SelectTrigger><SelectValue /></SelectTrigger>
   222→                <SelectContent><SelectItem value="CONTAINER">Container</SelectItem><SelectItem value="PREFAB">Prefab building</SelectItem></SelectContent>
   223→              </Select>
   224→            </div>
   225→            <div><Label>Size</Label><Input value={form.size} onChange={(e) => setForm({ ...form, size: e.target.value })} placeholder="e.g. 6 m, 12 m, 3x6 m" /></div>
   226→            <div className="sm:col-span-2"><Label>Description</Label><Input value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} placeholder="e.g. Converted container – site office" /></div>
   227→            <div>
   228→              <Label>Status</Label>
   229→              <Select value={form.status} onValueChange={(v) => setForm({ ...form, status: v })}>
   230→                <SelectTrigger><SelectValue /></SelectTrigger>
   231→                <SelectContent><SelectItem value="IN_STOCK">In stock</SelectItem><SelectItem value="CLIENT_SITE">At client</SelectItem><SelectItem value="MAINTENANCE">In maintenance</SelectItem><SelectItem value="RETIRED">Retired</SelectItem></SelectContent>
   232→              </Select>
   233→            </div>
   234→            <div><Label>Build date</Label><Input type="date" value={form.buildDate} onChange={(e) => setForm({ ...form, buildDate: e.target.value })} /></div>
   235→            <div className="sm:col-span-2 mt-1">
   236→              <p className="text-sm font-semibold mb-1.5 flex items-center gap-1.5"><Wrench className="h-4 w-4" /> Build team – who built what</p>
   237→              <div className="grid gap-3 sm:grid-cols-2">
   238→                <div><Label>Builder / fabricator</Label><Input value={form.builder} onChange={(e) => setForm({ ...form, builder: e.target.value })} /></div>
   239→                <div><Label>Electrician</Label><Input value={form.electrician} onChange={(e) => setForm({ ...form, electrician: e.target.value })} /></div>
   240→                <div><Label>Plumber</Label><Input value={form.plumber} onChange={(e) => setForm({ ...form, plumber: e.target.value })} /></div>
   241→                <div><Label>Painter</Label><Input value={form.painter} onChange={(e) => setForm({ ...form, painter: e.target.value })} /></div>
   242→                <div><Label>Glazier (windows)</Label><Input value={form.glazier} onChange={(e) => setForm({ ...form, glazier: e.target.value })} /></div>
   243→                <div><Label>Roofer</Label><Input value={form.roofer} onChange={(e) => setForm({ ...form, roofer: e.target.value })} /></div>
   244→                <div className="sm:col-span-2"><Label>Other trade / contractor</Label><Input value={form.otherTrade} onChange={(e) => setForm({ ...form, otherTrade: e.target.value })} placeholder="e.g. tiler, carpenter" /></div>
   245→              </div>
   246→            </div>
   247→            {form.status === 'CLIENT_SITE' && (
   248→              <>
   249→                <div><Label>Client</Label><Input value={form.client} onChange={(e) => setForm({ ...form, client: e.target.value })} /></div>
   250→                <div><Label>Site</Label><Input value={form.site} onChange={(e) => setForm({ ...form, site: e.target.value })} /></div>
   251→                <div><Label>Due back</Label><Input type="date" value={form.dueBack} onChange={(e) => setForm({ ...form, dueBack: e.target.value })} /></div>
   252→              </>
   253→            )}
   254→            <div className="sm:col-span-2"><Label>Notes</Label><Input value={form.notes} onChange={(e) => setForm({ ...form, notes: e.target.value })} /></div>
   255→          </div>
   256→          <div className="flex justify-end gap-2 mt-2">
   257→            <Button variant="outline" onClick={() => setEditOpen(false)}>Cancel</Button>
   258→            <Button onClick={saveUnit}>Save unit</Button>
   259→          </div>
   260→        </DialogContent>
   261→      </Dialog>
   262→
   263→      {/* Unit detail */}
   264→      <Dialog open={Boolean(detail)} onOpenChange={(v) => !v && setDetail(null)}>
   265→        <DialogContent className="max-w-3xl max-h-[92vh] overflow-y-auto">
   266→          {detail && (
   267→            <>
   268→              <DialogHeader>
   269→                <DialogTitle className="flex flex-wrap items-center gap-2">{detail.code} <StatusBadge status={detail.status} /></DialogTitle>
   270→              </DialogHeader>
   271→              <Tabs value={detailTab} onValueChange={setDetailTab}>
   272→                <TabsList className="flex-wrap h-auto [&>[data-slot=tabs-trigger]]:h-auto [&>[data-slot=tabs-trigger]]:flex-none">
   273→                  <TabsTrigger value="overview">Overview</TabsTrigger>
   274→                  <TabsTrigger value="contents">Contents ({detail.equipment.length})</TabsTrigger>
   275→                  <TabsTrigger value="maintenance">Maintenance ({(detail.schedules || []).length})</TabsTrigger>
   276→                  <TabsTrigger value="inspections">Inspections ({(detail.inspections || []).length})</TabsTrigger>
   277→                  <TabsTrigger value="history">History</TabsTrigger>
   278→                </TabsList>
   279→
   280→                <TabsContent value="overview" className="mt-3 space-y-3">
   281→                  <div className="grid gap-2 sm:grid-cols-2 text-sm">
   282→                    <p><span className="text-muted-foreground">Type / size:</span> {detail.type} {detail.size || ''}</p>
   283→                    <p><span className="text-muted-foreground">Serial:</span> {detail.serial || '-'}</p>
   284→                    <p><span className="text-muted-foreground">Built:</span> {fmtDate(detail.buildDate)}</p>
   285→                    <p><span className="text-muted-foreground">Client / site:</span> {detail.client || '-'}{detail.site ? ` / ${detail.site}` : ''}</p>
   286→                    {detail.outDate && <p><span className="text-muted-foreground">Out since:</span> {fmtDate(detail.outDate)}</p>}
   287→                    {detail.dueBack && <p><span className="text-muted-foreground">Due back:</span> {fmtDate(detail.dueBack)}</p>}
   288→                  </div>
   289→                  <div>
   290→                    <p className="text-sm font-semibold mb-1.5">Build team</p>
   291→                    <div className="grid gap-2 sm:grid-cols-3">
   292→                      {[['Builder', detail.builder], ['Electrician', detail.electrician], ['Plumber', detail.plumber], ['Painter', detail.painter], ['Glazier', detail.glazier], ['Roofer', detail.roofer], ['Other', detail.otherTrade]].map(([k, v]) => (
   293→                        <div key={k as string} className="rounded-md border p-2 text-sm"><span className="block text-xs text-muted-foreground">{k}</span>{(v as string) || '-'}</div>
   294→                      ))}
   295→                    </div>
   296→                  </div>
   297→                  <div className="flex flex-wrap gap-2 pt-1">
   298→                    <Button size="sm" variant="outline" onClick={() => generateSchedules(detail)} disabled={generating}><Wrench className="h-4 w-4 mr-1" /> Generate standard maintenance schedule</Button>
   299→                    <Button size="sm" variant="outline" onClick={() => downloadUnitHistoryReport(detail as never, 'Maintenance')}><FileDown className="h-4 w-4 mr-1" /> History report (PDF)</Button>
   300→                    <Button size="sm" variant="outline" onClick={() => openInsp(detail, 'QCP')}><Plus className="h-4 w-4 mr-1" /> QCP inspection</Button>
   301→                  </div>
   302→                  {detail.notes && <p className="text-sm text-muted-foreground">Notes: {detail.notes}</p>}
   303→                </TabsContent>
   304→
   305→                <TabsContent value="contents" className="mt-3">
   306→                  <div className="flex justify-between items-center mb-2">
   307→                    <p className="text-sm text-muted-foreground">Aircons, doors, windows and other contents of this unit.</p>
   308→                    <Button size="sm" onClick={() => setEqEdit({ kind: 'AIRCON', qty: 1 })}><Plus className="h-4 w-4 mr-1" /> Add item</Button>
   309→                  </div>
   310→                  <div className="rounded-lg border divide-y">
   311→                    {detail.equipment.length === 0 && <p className="p-3 text-sm text-muted-foreground">No contents recorded yet.</p>}
   312→                    {detail.equipment.map((e) => (
   313→                      <div key={e.id} className="flex items-center justify-between gap-2 p-2.5 text-sm">
   314→                        <div className="min-w-0">
   315→                          <p className="font-medium truncate">{e.name} {e.qty > 1 && <span className="text-muted-foreground">x{e.qty}</span>}</p>
   316→                          <p className="text-xs text-muted-foreground">
   317→                            {e.kind === 'AIRCON' && e.airconModel ? `${e.airconModel.brand} · service every ${e.airconModel.serviceIntervalMonths} mo` : e.kind}
   318→                            {e.serial ? ` · SN ${e.serial}` : ''} {e.installedDate ? `· fitted ${fmtDate(e.installedDate)}` : ''}
   319→                          </p>
   320→                        </div>
   321→                        <div className="flex gap-1 shrink-0">
   322→                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => setEqEdit({ ...e, installedDate: dateInput(e.installedDate) })}><Pencil className="h-4 w-4" /></Button>
   323→                          <Button variant="ghost" size="icon" className="h-8 w-8 text-red-600" onClick={() => delEquipment(e.id)}><Trash2 className="h-4 w-4" /></Button>
   324→                        </div>
   325→                      </div>
   326→                    ))}
   327→                  </div>
   328→                </TabsContent>
   329→
   330→                <TabsContent value="maintenance" className="mt-3">
   331→                  <div className="rounded-lg border divide-y max-h-96 overflow-y-auto">
   332→                    {(detail.schedules || []).length === 0 && <p className="p-3 text-sm text-muted-foreground">No schedules yet – use “Generate standard maintenance schedule”.</p>}
   333→                    {(detail.schedules || []).map((s) => (
   334→                      <div key={s.id} className="flex items-center justify-between gap-2 p-2.5 text-sm">
   335→                        <div><p className="font-medium">{s.name}</p><p className="text-xs text-muted-foreground">{s.type.name} · every {s.intervalMonths} months · {s.responsible || 'unassigned'}</p></div>
   336→                        <div className="text-right"><DueBadge date={s.nextDue} /><p className="text-xs text-muted-foreground mt-0.5">{fmtDate(s.nextDue)}</p></div>
   337→                      </div>
   338→                    ))}
   339→                  </div>
   340→                  <Button size="sm" variant="link" className="mt-2 px-0" onClick={() => onNavigate('maintenance')}>Open maintenance module →</Button>
   341→                </TabsContent>
   342→
   343→                <TabsContent value="inspections" className="mt-3">
   344→                  <div className="rounded-lg border divide-y max-h-96 overflow-y-auto">
   345→                    {(detail.inspections || []).length === 0 && <p className="p-3 text-sm text-muted-foreground">No inspections recorded yet.</p>}
   346→                    {(detail.inspections || []).map((i) => (
   347→                      <div key={i.id} className="flex items-center justify-between gap-2 p-2.5 text-sm">
   348→                        <div>
   349→                          <p className="font-medium">{i.type.replaceAll('_', ' ')} {i.type === 'QCP' && i.template ? `– ${i.template.name}` : ''}</p>
